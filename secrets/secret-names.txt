# Secret Manager secret ids for this deployment, one per line.
#
# scripts/bootstrap-gcp.sh creates each as an EMPTY container (no versions),
# so platform/ can grant access by id before any value exists. Values are
# added by hand with `gcloud secrets versions add` -- no infrastructure code
# ever reads one. Schemas: secrets/README.md.
#
# Per-stack secrets follow vaultwarden-<stack>-<name>; platform/ derives the
# grants from that convention, and platform/main_test.go fails if a granted
# name is missing here.

# Shared by both stacks. Read by Ansible (via the deployer SA).
vaultwarden-vaultwarden
vaultwarden-msmtp

# Per stack. Read by Ansible (via the deployer SA).
vaultwarden-disposable-restic
vaultwarden-disposable-healthchecks
vaultwarden-production-restic
vaultwarden-production-healthchecks

# Owner-only: the platform stack's provider credentials -- the Cloudflare
# token (R2 bucket admin) and the full-access Resend key (mail domain).
# Deliberately NOT granted to the deployer SA; platform/main_test.go checks.
vaultwarden-cloudflare-api-token
vaultwarden-resend-api-key
